# React/TypeScript Implementation

This guide covers integrating the TX State Machine into React applications with TypeScript, including custom hooks, state management, and UI patterns.

## Overview

The integration consists of:
1. **API client** - Functions to call Gateway endpoints
2. **Custom hook** - `useTransaction` for managing the full lifecycle
3. **UI components** - Transaction status display and user feedback

---

## API Client

Create a typed API client for the TX State Machine endpoints:

```typescript
// lib/api/tx-state-machine.ts

const API_BASE = process.env.NEXT_PUBLIC_ANDAMIO_GATEWAY_URL || 'https://dev.api.andamio.io';

export type TxState = 'pending' | 'confirmed' | 'updated' | 'failed' | 'expired';

export interface TxStatus {
  tx_hash: string;
  tx_type: string;
  state: TxState;
  created_at: string;
  confirmed_at?: string;
  updated_at?: string;
  last_error?: string;
}

export interface RegisterTxRequest {
  tx_hash: string;
  tx_type: string;
  metadata?: Record<string, unknown>;
}

export async function registerTransaction(
  request: RegisterTxRequest,
  auth: { apiKey: string; jwt: string }
): Promise<{ success: boolean }> {
  const response = await fetch(`${API_BASE}/api/v2/tx/register`, {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      'X-API-Key': auth.apiKey,
      'Authorization': `Bearer ${auth.jwt}`,
    },
    body: JSON.stringify(request),
  });

  if (!response.ok) {
    throw new Error(`Registration failed: ${response.statusText}`);
  }

  return response.json();
}

export async function getTransactionStatus(
  txHash: string,
  auth: { apiKey: string; jwt: string }
): Promise<TxStatus> {
  const response = await fetch(`${API_BASE}/api/v2/tx/status/${txHash}`, {
    headers: {
      'X-API-Key': auth.apiKey,
      'Authorization': `Bearer ${auth.jwt}`,
    },
  });

  if (!response.ok) {
    throw new Error(`Status check failed: ${response.statusText}`);
  }

  return response.json();
}

export async function getPendingTransactions(
  auth: { apiKey: string; jwt: string }
): Promise<TxStatus[]> {
  const response = await fetch(`${API_BASE}/api/v2/tx/pending`, {
    headers: {
      'X-API-Key': auth.apiKey,
      'Authorization': `Bearer ${auth.jwt}`,
    },
  });

  if (!response.ok) {
    throw new Error(`Failed to fetch pending transactions: ${response.statusText}`);
  }

  return response.json();
}
```

## useTransaction Hook

This hook manages the complete transaction lifecycle:

```typescript
// hooks/useTransaction.ts

import { useState, useCallback, useRef } from 'react';
import {
  registerTransaction,
  getTransactionStatus,
  TxState,
  TxStatus
} from '@/lib/api/tx-state-machine';

export type TransactionPhase =
  | 'idle'
  | 'building'
  | 'signing'
  | 'submitting'
  | 'registering'
  | 'confirming'
  | 'complete'
  | 'error';

interface UseTransactionOptions {
  apiKey: string;
  jwt: string;
  onSuccess?: (status: TxStatus) => void;
  onError?: (error: Error) => void;
  pollInterval?: number;  // milliseconds, default 10000
  pollDelay?: number;     // initial delay before polling, default 30000
}

interface UseTransactionReturn {
  phase: TransactionPhase;
  txHash: string | null;
  txStatus: TxStatus | null;
  error: Error | null;
  execute: (params: {
    buildTx: () => Promise<string>;  // Returns unsigned CBOR
    signTx: (cbor: string) => Promise<string>;  // Returns signed TX
    submitTx: (signedTx: string) => Promise<string>;  // Returns tx_hash
    txType: string;
    metadata?: Record<string, unknown>;
  }) => Promise<void>;
  reset: () => void;
}

export function useTransaction(options: UseTransactionOptions): UseTransactionReturn {
  const {
    apiKey,
    jwt,
    onSuccess,
    onError,
    pollInterval = 10000,
    pollDelay = 30000,
  } = options;

  const [phase, setPhase] = useState<TransactionPhase>('idle');
  const [txHash, setTxHash] = useState<string | null>(null);
  const [txStatus, setTxStatus] = useState<TxStatus | null>(null);
  const [error, setError] = useState<Error | null>(null);

  const pollRef = useRef<NodeJS.Timeout | null>(null);

  const stopPolling = useCallback(() => {
    if (pollRef.current) {
      clearTimeout(pollRef.current);
      pollRef.current = null;
    }
  }, []);

  const pollStatus = useCallback(async (hash: string) => {
    try {
      const status = await getTransactionStatus(hash, { apiKey, jwt });
      setTxStatus(status);

      if (status.state === 'updated') {
        setPhase('complete');
        stopPolling();
        onSuccess?.(status);
      } else if (status.state === 'failed' || status.state === 'expired') {
        const err = new Error(status.last_error || `Transaction ${status.state}`);
        setError(err);
        setPhase('error');
        stopPolling();
        onError?.(err);
      } else {
        // Still pending or confirmed, keep polling
        pollRef.current = setTimeout(() => pollStatus(hash), pollInterval);
      }
    } catch (err) {
      const error = err instanceof Error ? err : new Error('Unknown error');
      setError(error);
      setPhase('error');
      stopPolling();
      onError?.(error);
    }
  }, [apiKey, jwt, pollInterval, onSuccess, onError, stopPolling]);

  const execute = useCallback(async (params: {
    buildTx: () => Promise<string>;
    signTx: (cbor: string) => Promise<string>;
    submitTx: (signedTx: string) => Promise<string>;
    txType: string;
    metadata?: Record<string, unknown>;
  }) => {
    const { buildTx, signTx, submitTx, txType, metadata } = params;

    try {
      // Reset state
      setError(null);
      setTxHash(null);
      setTxStatus(null);
      stopPolling();

      // 1. Build
      setPhase('building');
      const cbor = await buildTx();

      // 2. Sign
      setPhase('signing');
      const signedTx = await signTx(cbor);

      // 3. Submit
      setPhase('submitting');
      const hash = await submitTx(signedTx);
      setTxHash(hash);

      // 4. Register
      setPhase('registering');
      await registerTransaction(
        { tx_hash: hash, tx_type: txType, metadata },
        { apiKey, jwt }
      );

      // 5. Start polling after delay
      setPhase('confirming');
      pollRef.current = setTimeout(() => pollStatus(hash), pollDelay);

    } catch (err) {
      const error = err instanceof Error ? err : new Error('Unknown error');
      setError(error);
      setPhase('error');
      onError?.(error);
    }
  }, [apiKey, jwt, pollDelay, pollStatus, stopPolling, onError]);

  const reset = useCallback(() => {
    stopPolling();
    setPhase('idle');
    setTxHash(null);
    setTxStatus(null);
    setError(null);
  }, [stopPolling]);

  return {
    phase,
    txHash,
    txStatus,
    error,
    execute,
    reset,
  };
}
```

## Usage Example

Here's a complete example of submitting an assignment:

```tsx
// components/SubmitAssignmentButton.tsx

import { useTransaction } from '@/hooks/useTransaction';
import { useWallet } from '@/hooks/useWallet';  // Your wallet hook
import { useAuth } from '@/hooks/useAuth';      // Your auth hook
import { toast } from 'sonner';

interface Props {
  courseId: string;
  moduleHash: string;
  evidenceLink: string;
}

export function SubmitAssignmentButton({ courseId, moduleHash, evidenceLink }: Props) {
  const { wallet, address } = useWallet();
  const { apiKey, jwt } = useAuth();

  const { phase, txHash, error, execute, reset } = useTransaction({
    apiKey,
    jwt,
    onSuccess: (status) => {
      toast.success('Assignment submitted successfully!');
      // Optionally refresh data
    },
    onError: (error) => {
      toast.error(`Transaction failed: ${error.message}`);
    },
  });

  const handleSubmit = async () => {
    await execute({
      // 1. Build transaction
      buildTx: async () => {
        const response = await fetch('/api/v2/tx/course/student/assignment/commit', {
          method: 'POST',
          headers: {
            'Content-Type': 'application/json',
            'X-API-Key': apiKey,
            'Authorization': `Bearer ${jwt}`,
          },
          body: JSON.stringify({
            alias: address,
            course_id: courseId,
            slt_hash: moduleHash,
            assignment_info: { evidence_link: evidenceLink },
          }),
        });
        const data = await response.json();
        return data.cbor;
      },

      // 2. Sign with wallet
      signTx: (cbor) => wallet.signTx(cbor, true),

      // 3. Submit to network
      submitTx: (signedTx) => wallet.submitTx(signedTx),

      // 4. Transaction type for registration
      txType: 'assignment_submit',
    });
  };

  const isLoading = phase !== 'idle' && phase !== 'complete' && phase !== 'error';

  return (
    <div>
      <button
        onClick={handleSubmit}
        disabled={isLoading}
        className="btn btn-primary"
      >
        {isLoading ? <PhaseIndicator phase={phase} /> : 'Submit Assignment'}
      </button>

      {txHash && (
        <p className="text-sm text-muted-foreground mt-2">
          TX: <code>{txHash.slice(0, 16)}...</code>
        </p>
      )}

      {error && (
        <p className="text-sm text-destructive mt-2">
          {error.message}
          <button onClick={reset} className="ml-2 underline">Try again</button>
        </p>
      )}
    </div>
  );
}

function PhaseIndicator({ phase }: { phase: string }) {
  const labels: Record<string, string> = {
    building: 'Building transaction...',
    signing: 'Please sign in your wallet...',
    submitting: 'Submitting to network...',
    registering: 'Registering transaction...',
    confirming: 'Waiting for confirmation...',
  };
  return <span>{labels[phase] || 'Processing...'}</span>;
}
```

## Transaction Status Component

Display pending transactions for the current user:

```tsx
// components/PendingTransactions.tsx

import { useEffect, useState } from 'react';
import { getPendingTransactions, TxStatus } from '@/lib/api/tx-state-machine';
import { useAuth } from '@/hooks/useAuth';

export function PendingTransactions() {
  const { apiKey, jwt } = useAuth();
  const [transactions, setTransactions] = useState<TxStatus[]>([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    const fetchPending = async () => {
      try {
        const pending = await getPendingTransactions({ apiKey, jwt });
        setTransactions(pending);
      } catch (error) {
        console.error('Failed to fetch pending transactions:', error);
      } finally {
        setLoading(false);
      }
    };

    fetchPending();
    const interval = setInterval(fetchPending, 30000);
    return () => clearInterval(interval);
  }, [apiKey, jwt]);

  if (loading) return <div>Loading...</div>;
  if (transactions.length === 0) return null;

  return (
    <div className="p-4 bg-muted rounded-lg">
      <h3 className="font-medium mb-2">Pending Transactions</h3>
      <ul className="space-y-2">
        {transactions.map((tx) => (
          <li key={tx.tx_hash} className="flex items-center gap-2">
            <StatusBadge state={tx.state} />
            <span className="text-sm">{tx.tx_type}</span>
            <code className="text-xs text-muted-foreground">
              {tx.tx_hash.slice(0, 12)}...
            </code>
          </li>
        ))}
      </ul>
    </div>
  );
}

function StatusBadge({ state }: { state: string }) {
  const colors: Record<string, string> = {
    pending: 'bg-yellow-100 text-yellow-800',
    confirmed: 'bg-blue-100 text-blue-800',
    updated: 'bg-green-100 text-green-800',
    failed: 'bg-red-100 text-red-800',
    expired: 'bg-gray-100 text-gray-800',
  };

  return (
    <span className={`px-2 py-0.5 rounded text-xs ${colors[state] || ''}`}>
      {state}
    </span>
  );
}
```

## Error Handling

<Callout type="warn">
Always handle all terminal states: `updated`, `failed`, and `expired`.
</Callout>

Common errors and solutions:

| Error | Cause | Solution |
|-------|-------|----------|
| `invalid tx_type` | TX type not in valid list | Check `/tx/types` for valid values |
| `tx_hash must be 64 characters` | Invalid transaction hash | Ensure hash from wallet is correct |
| `Transaction not found` | TX expired or never registered | Register immediately after submit |
| `state: failed` | Error during confirmation or DB update | Check `last_error` for details |
| `state: expired` | TX not confirmed within 2 hours | Rebuild and resubmit |

---

## Best Practices

1. **Register immediately after submit** - Don't wait or do other operations between submit and register.

2. **Use appropriate poll timing** - 30 second initial delay, 10-15 second intervals.

3. **Clean up on unmount** - Stop polling when components unmount to prevent memory leaks.

4. **Show transaction hash early** - Display the hash immediately after submit so users can track externally if needed.

5. **Provide retry capability** - Allow users to reset and try again on failure.

6. **Consider optimistic updates** - For better UX, update local state optimistically while waiting for confirmation.
